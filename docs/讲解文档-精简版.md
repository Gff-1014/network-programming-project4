## 二、TCP 分帧：先解决消息边界

TCP 是字节流，一次 `recv` 可能包含两条消息，也可能只包含半条消息。项目使用“4 字节大端长度 + JSON 消息体”，接收端先精确读 4 字节长度，再按该长度精确读消息体。

```python
# server/framing.py:18
HEADER = struct.Struct(">I")


class FrameError(Exception):
    """帧不合法。code 直接作为 error 帧的错误码。"""

    def __init__(self, code: str, message: str, fatal: bool = False) -> None:
        super().__init__(f"{code}: {message}")
        self.code = code
        self.message = message
        # fatal=True 表示字节流已经错位，只能断开
        self.fatal = fatal


def encode_frame(payload: dict) -> bytes:
    body = json.dumps(
        payload,
        ensure_ascii=False,
        separators=(",", ":"),
    ).encode("utf-8")
    return HEADER.pack(len(body)) + body


async def read_frame(reader: asyncio.StreamReader, max_bytes: int) -> dict:
    header = await reader.readexactly(HEADER.size)
    (length,) = HEADER.unpack(header)

    if length == 0:
        raise FrameError("BAD_FRAME", "帧长度为 0")
    if length > max_bytes:
        raise FrameError(
            "BAD_FRAME",
            f"帧长度 {length} 超过上限 {max_bytes}",
            fatal=True,
        )

    body = await reader.readexactly(length)
    try:
        payload = json.loads(body.decode("utf-8"))
    except UnicodeDecodeError:
        raise FrameError("BAD_FRAME", "帧不是合法的 UTF-8") from None
    except json.JSONDecodeError:
        raise FrameError("BAD_FRAME", "帧不是合法的 JSON") from None

    if not isinstance(payload, dict):
        raise FrameError("BAD_FRAME", "消息必须是 JSON 对象")
    return payload
```

关键点：

- `readexactly(4)` 解决长度头本身可能分多次到达的问题。
- `readexactly(length)` 解决半包，并保证下一条消息从正确位置开始。
- 长度超限时消息体没有读走，字节流已错位，所以 `fatal=True`，必须断开。
- 对应测试：`tests/test_framing.py`，覆盖半包、粘包、长度为 0、长度超限和非法 JSON。

```python
# server/session.py:54
async def run(self) -> None:
    self._writer_task = asyncio.create_task(self._writer_loop())
    self.hub.add_session(self)
    try:
        first = await self._handshake()
        if first is None:
            return
        await self._dispatch(first)

        while not self.closed:
            try:
                payload = await read_frame(
                    self.reader,
                    self.cfg.max_frame_bytes,
                )
            except FrameError as exc:
                # 只有字节流仍然对齐时才能继续
                if not self._reject_bad_frame(exc):
                    break
                continue

            self.last_seen_ms = protocol.now_ms()
            await self._dispatch(payload)

    except FrameError as exc:
        self._reject_bad_frame(exc)
    except asyncio.TimeoutError:
        self.send(error(
            ErrCode.HANDSHAKE_TIMEOUT,
            "未在限时内完成登录",
        ))
    except asyncio.IncompleteReadError:
        pass
    except (ConnectionError, OSError) as exc:
        logger.info("[%s] 连接中断: %s", self.conn_id, exc)
    except Exception:
        logger.exception("[%s] 会话异常", self.conn_id)
    finally:
        await self.close(self.close_reason or "session end")
```

```python
# server/session.py:145
async def _writer_loop(self) -> None:
    try:
        while True:
            data = await self.queue.get()
            self._queue_bytes = max(0, self._queue_bytes - len(data))
            try:
                # 只阻塞当前连接的写协程
                self.writer.write(data)
                await self.writer.drain()
            finally:
                self.queue.task_done()
    except (asyncio.CancelledError, ConnectionError, OSError):
        pass


# server/session.py:187
def send(self, payload: dict, droppable: bool = False) -> bool:
    if self.closed or self.slow_consumer:
        return False

    data = encode_frame(payload)
    if (
        self.queue.full()
        or self._queue_bytes + len(data) > self.cfg.send_queue_max_bytes
    ):
        self._on_drop(droppable)
        return False

    self._queue_bytes += len(data)
    self.queue.put_nowait(data)
    self.queue_high_water = max(
        self.queue_high_water,
        self.queue.qsize(),
    )
    return True
```

慢客户端判定：

```python
# server/session.py:215
def _on_drop(self, droppable: bool) -> None:
    self.dropped += 1
    self.hub.dropped_total += 1
    now = protocol.now_ms()
    self.drop_times.append(now)

    while (
        self.drop_times
        and now - self.drop_times[0] > self.cfg.slow_consumer_window_ms
    ):
        self.drop_times.popleft()

    if not droppable:
        logger.warning(
            "[%s] 队列已满，丢弃必达消息（累计 %d）",
            self.conn_id,
            self.dropped,
        )
    if len(self.drop_times) >= self.cfg.slow_consumer_drop_limit:
        self._mark_slow_consumer()


# server/session.py:228
def _mark_slow_consumer(self) -> None:
    if self.slow_consumer or self.closed:
        return

    self.slow_consumer = True
    self.hub.slow_consumer_disconnects += 1
    logger.warning(
        "[%s] 判定慢客户端：窗口内丢弃 %d 条，断开连接",
        self.conn_id,
        len(self.drop_times),
    )

    while not self.queue.empty():
        try:
            self.queue.get_nowait()
            self.queue.task_done()
        except asyncio.QueueEmpty:
            break

    self._queue_bytes = 0
    try:
        self.queue.put_nowait(encode_frame(error(
            ErrCode.SLOW_CONSUMER,
            "发送队列持续积压，连接将被断开",
        )))
    except asyncio.QueueFull:
        pass

    asyncio.get_running_loop().create_task(self._close_later(0.3))
```

核心结论：

- 每条连接独立队列，默认同时限制条数和字节数。
- 慢客户端只会卡住自己的 `_writer_loop()`。
- 窗口内持续丢弃后，服务端清空该连接积压、返回 `SLOW_CONSUMER` 并断开。
- 其他客户端继续使用自己的队列和写协程。

## 四、心跳与静默清理

```python
# server/main.py:27
async def heartbeat_sweeper(hub: Hub) -> None:
    cfg = hub.cfg
    while True:
        await asyncio.sleep(cfg.sweep_interval_ms / 1000)
        now = protocol.now_ms()

        for session in list(hub.sessions.values()):
            if session.closed:
                continue

            idle = now - session.last_seen_ms
            if (
                idle >= cfg.ping_interval_ms
                and now - session.last_ping_ms >= cfg.ping_interval_ms
            ):
                session.ping()

            if idle >= cfg.idle_timeout_ms:
                logger.info(
                    "[%s] 静默 %d ms，判定失效",
                    session.conn_id,
                    idle,
                )
                await session.close_for_idle()

        hub.sweep_reservations()
        for room in hub.rooms.values():
            if room.table is not None:
                await room.table.tick(now)
```

```python
async def close_for_idle(self) -> None:
    self.hub.idle_timeouts += 1
    self.send_control(error(
        ErrCode.SESSION_EXPIRED,
        "长时间无响应，连接已断开",
    ))
    await self.close("idle timeout", keep_reservation=True)
```

关键点：

- 收到任意合法帧都会刷新 `last_seen_ms`。
- 静默超过 `ping_interval_ms` 后服务端发送 `ping`。
- 超过 `idle_timeout_ms` 后判定失效并清理。
- `ping` 走控制通道，不进入业务队列，避免被慢客户端积压挡住。

```python
# server/session.py:102
async def _dispatch(self, payload: dict) -> None:
    msg_type = payload.get("type")
    request_id = payload.get("request_id")

    if not isinstance(msg_type, str):
        self.bad_frames += 1
        self.send(error(
            ErrCode.BAD_MESSAGE,
            "缺少 type 字段",
            request_id,
        ))
        return

    handler = protocol.get_handler(msg_type)
    if handler is None:
        self.unknown_types += 1
        self.send(error(
            ErrCode.UNKNOWN_TYPE,
            f"未知消息类型 {msg_type}",
            request_id,
            supported=protocol.SUPPORTED_TYPES,
        ))
        if self.unknown_types >= self.cfg.bad_frame_limit:
            await self.close("too many unknown types")
        return

    data = payload.get("data") or {}
    if not isinstance(data, dict):
        data = {}

    try:
        await handler(self, data, request_id)
    except Exception:
        logger.exception(
            "[%s] 处理 %s 时异常",
            self.conn_id,
            msg_type,
        )
        self.send(error(
            ErrCode.INTERNAL,
            f"处理 {msg_type} 时内部错误",
            request_id,
        ))
```

```python
def make(
    msg_type: str,
    data: dict | None = None,
    request_id: str | None = None,
) -> dict:
    payload = {"type": msg_type, "data": data or {}}
    if request_id:
        payload["request_id"] = request_id
    payload["ts_ms"] = now_ms()
    return payload


def error(
    code: str,
    message: str,
    request_id: str | None = None,
    **extra,
) -> dict:
    data = {"code": code, "message": message}
    data.update(extra)
    return make("error", data, request_id)


Handler = Callable[..., Awaitable[None]]
_HANDLERS: dict[str, Handler] = {}


def register(msg_type: str):
    def decorator(func: Handler) -> Handler:
        _HANDLERS[msg_type] = func
        return func
    return decorator


def get_handler(msg_type: str) -> Handler | None:
    return _HANDLERS.get(msg_type)
```

房间只保存自己的成员：

```python
# server/hub.py:60
class Room:
    def __init__(self, name: str, kind: str = "chat") -> None:
        self.name = name
        self.kind = kind
        self.members: dict[str, "Session"] = {}
        self.owner: str | None = None
        self.table: Table | None = None

    def add(self, session: "Session") -> None:
        self.members[session.conn_id] = session

    def remove(self, session: "Session") -> None:
        self.members.pop(session.conn_id, None)

    def nicknames(self) -> list[str]:
        return sorted(
            s.nickname
            for s in self.members.values()
            if s.nickname
        )

    def broadcast(
        self,
        payload: dict,
        droppable: bool = False,
        exclude: "Session | None" = None,
    ) -> int:
        sent = 0
        for member in list(self.members.values()):
            if member is exclude:
                continue
            if member.send(payload, droppable=droppable):
                sent += 1
        return sent
```

`Room.broadcast()` 只遍历本房间成员，因此房间隔离由服务端保证。

```python
# game/engine.py:453
def _cached(
    self,
    player: str,
    request_id: str | None,
) -> Action | None:
    if not request_id:
        return None

    key = (player, request_id)
    hit = self._requests.get(key)
    if hit is None:
        return None

    self._requests.move_to_end(key)
    return Action(
        ok=hit.ok,
        code=hit.code,
        detail=hit.detail,
        duplicate=True,
    )


def _remember(
    self,
    player: str,
    request_id: str | None,
    action: Action,
) -> None:
    if not request_id:
        return

    key = (player, request_id)
    self._requests[key] = action
    self._requests.move_to_end(key)

    while len(self._requests) > self.cfg.request_cache_size:
        self._requests.popitem(last=False)
```

## 八、客户端单读者与请求响应分流

同一条 TCP 连接只能有一个真正的读循环：

```python
# client/net.py:49
async def _pump(self) -> None:
    try:
        while True:
            frame = await read_frame(
                self.reader,
                self.max_frame_bytes,
            )

            if frame.get("type") == "ping" and self.auto_pong:
                self.writer.write(encode_frame({
                    "type": "pong",
                    "data": {},
                }))
                await self.writer.drain()

            request_id = frame.get("request_id")
            waiter = (
                self._waiters.pop(request_id, None)
                if request_id
                else None
            )

            if waiter is not None and not waiter.done():
                waiter.set_result(frame)
            else:
                self._inbox.put_nowait(frame)

    except asyncio.CancelledError:
        raise
    except Exception:
        self._mark_closed()


# client/net.py:134
async def request(
    self,
    msg_type: str,
    data: dict | None = None,
    timeout: float = 5.0,
) -> dict:
    rid = f"c{next(self._ids)}"
    waiter = asyncio.get_running_loop().create_future()
    self._waiters[rid] = waiter
    await self.send(msg_type, data, request_id=rid)

    try:
        frame = await asyncio.wait_for(waiter, timeout)
    except asyncio.TimeoutError:
        self._waiters.pop(rid, None)
        raise asyncio.TimeoutError(
            f"{msg_type} 等待响应超时"
        ) from None

    return self._as_response(frame)
```

关键点：

- 带 `request_id` 且在等待表中有对应 future 的帧，是某个请求的响应。
- 其他帧进入收件箱，作为服务端主动推送。
- 这样请求响应和房间广播不会互相抢包，也不会出现多个协程并发读流。

## 九、网页版：浏览器通过 WebSocket 复用 TCP 服务端

网页版没有另写一套业务服务端。浏览器只负责界面，`web_gateway.py` 为每个浏览器
标签页创建一个 `ClientSession`，由它连接原来的 TCP 服务端。因此网页、桌面客户端
和机器人共享同一套账号、房间、私信、观战和牌桌状态。

### 9.1 浏览器到服务端的桥

每个 WebSocket 连接对应一个 `BrowserBridge`：

```python
# web_gateway.py:39
class BrowserBridge:
    """一条浏览器 WebSocket 对应的 TCP 客户端。"""

    def __init__(self, backend_host: str, backend_port: int) -> None:
        self.backend_host = backend_host
        self.backend_port = backend_port
        self.events: asyncio.Queue[dict] = asyncio.Queue()
        self.session = ClientSession(on_event=self._on_event)
        self.closed = False

    def _on_event(self, kind: str, data: dict) -> None:
        if not self.closed:
            self.events.put_nowait({
                "type": "event",
                "kind": kind,
                "data": data,
            })

    async def _send_loop(self, ws: web.WebSocketResponse) -> None:
        while not self.closed:
            message = await self.events.get()
            try:
                await ws.send_json(message)
            except (ConnectionError, RuntimeError):
                return

    async def ensure_connected(self) -> None:
        conn = self.session.conn
        if conn is None or self.session._closed or conn.closed:
            await self.session.connect(self.backend_host, self.backend_port)
```

`ClientSession` 的事件先进入队列，再由单独的发送协程写成 WebSocket JSON。
这样浏览器慢慢读取时，不会阻塞 TCP 客户端的收包协程。

### 9.2 WebSocket 生命周期

```python
# web_gateway.py:258
async def run(self, ws: web.WebSocketResponse) -> None:
    send_task = asyncio.create_task(self._send_loop(ws))
    try:
        async for message in ws:
            if message.type == WSMsgType.TEXT:
                payload = json.loads(message.data)
                await self.handle(payload)
            elif message.type == WSMsgType.ERROR:
                break
    finally:
        self.closed = True
        send_task.cancel()
        await asyncio.gather(send_task, return_exceptions=True)
        await self.session.close()


# web_gateway.py:284
async def ws_handler(request: web.Request) -> web.WebSocketResponse:
    bridge = BrowserBridge(
        request.app["backend_host"],
        request.app["backend_port"],
    )
    ws = web.WebSocketResponse(heartbeat=20.0, max_msg_size=64 * 1024)
    await ws.prepare(request)
    await bridge.send({"type": "hello", "data": {"websocket": True}})

```

浏览器关闭页面时，`finally` 会关闭该会话。正常关闭会执行 `logout`，释放账号和
牌桌座位；异常掉线仍由原服务端保留 60 秒 token 重连预约。

### 9.3 网页命令到现有客户端方法的映射

`BrowserBridge.handle()` 不重新解析 TCP 协议，而是把网页命令转给已有的
`ClientSession` 方法：

| 网页命令 | 调用 |
| --- | --- |
| `login` / `register` | `session.login()` / `session.register()` |
| `join_room` / `leave_room` | `session.join_room()` / `session.leave_room()` |
| `room_msg` | `session.say()` |
| `private_open` / `private_msg` | `session.open_private()` / `session.send_private()` |
| `create_group` / `delete_group` | `session.create_group()` / `session.delete_group()` |
| `ready` / `bid` / `pass` | `session.ready()` / `session.bid()` / `session.pass_turn()` |
| `play` | `session.play([parse_card(code) ...])` |
| `set_auto` | `session.set_auto()` |

核心分发代码：

```python
# web_gateway.py:108
async def handle(self, message: dict) -> None:
    command = message.get("type")
    data = message.get("data") or {}

    if command in ("login", "register"):
        await self._handle_auth(command, data)
        return

    if command == "logout":
        await self.session.logout()
        await self.send_snapshot()
        return

    if command == "join_room":
        await self.session.join_room(_text(data.get("room")))
    elif command == "room_msg":
        await self.session.say(_text(data.get("text")))
    elif command == "private_open":
        await self.session.open_private(_text(data.get("peer")))
    elif command == "private_msg":
        await self.session.send_private(_text(data.get("text")))
    elif command == "play":
        await self.session.play([
            parse_card(item) for item in data.get("cards", [])
        ])
    elif command == "set_auto":
        await self.session.set_auto(bool(data.get("enabled", True)))
    elif command == "request_state":
        await self.session.request_state()

    await self.send({"type": "ack", "command": command})
    await self.send_snapshot()
```

完整版本还包含参数校验、错误码转换、刷新房间/用户/会话列表以及 `ready`、`bid`、
`pass` 等分支，见 `web_gateway.py`。

### 9.4 网页端的 WebSocket 代码

```javascript
// web/static/app.js:65
function connectSocket() {
  const protocol = location.protocol === "https:" ? "wss" : "ws";
  const socket = new WebSocket(`${protocol}://${location.host}/ws`);
  state.socket = socket;

  socket.onopen = () => {
    state.connected = true;
    setAuthStatus("已连接服务端");
    send("hello");
  };

  socket.onmessage = (event) => {
    handleMessage(JSON.parse(event.data));
  };

  socket.onclose = () => {
    state.connected = false;
    reconnectTimer = setTimeout(connectSocket, 1500);
  };
}

function send(type, data = {}) {
  if (!state.socket || state.socket.readyState !== WebSocket.OPEN) {
    showToast("连接尚未建立，请稍后重试", true);
    return false;
  }
  state.socket.send(JSON.stringify({ type, data }));
  return true;
}

function handleMessage(message) {
  if (message.type === "snapshot") {
    applySnapshot(message.data || {});
  } else if (message.type === "event") {
    handleEvent(message.kind, message.data || {});
  } else if (message.type === "error") {
    showToast(message.message || message.code || "操作失败", true);
  }
}
```

`snapshot` 用于同步完整状态，`event` 用于增量更新，`error` 用于显示操作失败原因。

### 9.8 扫码入群与群功能

群聊新增四个功能：

- **扫码入群**：群工具生成 `/join/<invite_token>` 二维码，手机扫码登录后发送
  `join_invite` 入群申请，群主审批通过后才加入；
- **群成员可见性**：群聊不在非成员房间列表中，非成员即使知道内部键也不能直接加入；
- **自定义群名**：群主可以修改显示群名，显示名允许重复，内部仍使用唯一房间键和
  `group_id`；
- **群公告**：`Room.announcement` 保存纯文本公告，群主更新后服务器广播
  `announcement_updated`；
- **群投票**：`GroupPoll` 保存标题、选项、截止时间和每个账号的一票，服务端统一
  校验并广播 `group_state`；
- **群文件**：文件内容保存在 `data/uploads/`，TCP 协议只广播 `group_file_added`
  元数据，网页通过 `/api/attachments/{file_id}` 预览或下载。

协议入口在 `server/protocol.py`，群状态在 `server/hub.py`，网页上传/下载和二维码
接口在 `web_gateway.py`，界面入口在 `web/static/app.js`。
